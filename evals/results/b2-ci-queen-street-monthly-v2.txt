```python
import pandas as pd

file = '/mnt/data/assistant-HMX9DEaWdLgVDrtKen5KEi-atlas_sales.xlsx'
df = pd.read_excel(file, sheet_name='Orders')
df['order_date'] = pd.to_datetime(df['order_date'])

qs = df[(df['store'] == 'Toronto - Queen Street') & (df['order_date'].dt.year == 2025)].copy()
qs['month'] = qs['order_date'].dt.to_period('M').dt.to_timestamp()

monthly = qs.groupby('month', as_index=False)['amount'].sum()
print(monthly)
```
